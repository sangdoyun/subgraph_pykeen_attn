"""
Temporal Pattern Detection using Autocorrelation
Dependencies: pip install pandas numpy statsmodels scipy matplotlib
"""
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from statsmodels.tsa.stattools import acf
from scipy.signal import find_peaks

def generate_synthetic_data(days=90):
    """Generates synthetic hourly event data over a specified number of days."""
    print(f"--- Generating {days} days of hourly data ---")
    
    # Create an hourly time series
    dates = pd.date_range(start='2026-01-01', periods=24*days, freq='h')
    df = pd.DataFrame({'timestamp': dates})
    
    # 1. Inject Daily Pattern: Call Mom at 6 PM (18:00)
    # We add a random factor (< 0.8) so it happens 80% of the time (not perfectly rigid)
    df['call_mom'] = np.where(
        (df['timestamp'].dt.hour == 18) & (np.random.rand(len(df)) < 0.8), 
        1, 0
    )
    
    # 2. Inject Weekly Pattern: Saturday Morning Activity at 10 AM
    # dayofweek == 5 is Saturday
    df['weekly_activity'] = np.where(
        (df['timestamp'].dt.dayofweek == 5) & (df['timestamp'].dt.hour == 10) & (np.random.rand(len(df)) < 0.9), 
        1, 0
    )
    
    # 3. Add Random Background Noise (random app usage, messages, etc.)
    df['noise'] = np.random.binomial(n=1, p=0.03, size=len(df)) # 3% chance of random event in any hour
    
    # Combine signals into a single "engagement" metric
    df['total_events'] = df['call_mom'] + df['weekly_activity'] + df['noise']
    
    return df

def detect_temporal_patterns(df, column_name, max_lag_hours=200):
    """
    Uses Autocorrelation (ACF) to find periodic patterns.
    max_lag_hours=200 covers a full week (168 hours) + extra buffer.
    """
    print("\n--- Running Autocorrelation Analysis ---")
    
    # Calculate Autocorrelation Function (ACF)
    # fft=True speeds up the calculation for large datasets
    lag_acf = acf(df[column_name], nlags=max_lag_hours, fft=True)
    
    # Find peaks in the ACF array
    # distance=12 ensures we don't get multiple peaks right next to each other
    # height=0.1 filters out very weak correlations (noise)
    peaks, properties = find_peaks(lag_acf, height=0.1, distance=12)
    
    print("Detected Autocorrelation Peaks (Lags in Hours):", peaks)
    
    # Interpret the peaks
    detected_patterns = []
    for peak in peaks:
        if peak == 0:
            continue # Lag 0 is always 1.0 (correlation with itself), ignore it.
            
        # Check if the peak is around 24 hours (Daily)
        if 22 <= peak <= 26:
            detected_patterns.append(f"Daily Pattern detected (Lag: {peak} hours)")
            
        # Check if the peak is around 168 hours (Weekly)
        elif 165 <= peak <= 171:
            detected_patterns.append(f"Weekly Pattern detected (Lag: {peak} hours)")
            
        # Check for multiples (e.g., 48 hours is just a harmonic of daily)
        elif peak % 24 <= 2 or peak % 24 >= 22:
            pass # Ignore harmonics of daily patterns to reduce spam
        else:
            detected_patterns.append(f"Unknown Periodic Pattern (Lag: {peak} hours)")
            
    return lag_acf, detected_patterns

def main():
    # 1. Generate Data
    df = generate_synthetic_data(days=90)
    
    # 2. Analyze
    acf_values, patterns = detect_temporal_patterns(df, 'total_events')
    
    # 3. Print Results
    print("\n--- Detection Results ---")
    if not patterns:
        print("No dominant temporal patterns found.")
    for p in patterns:
        print(f"✅ {p}")
        
    # 4. Visualization (Optional, requires Matplotlib)
    plt.figure(figsize=(10, 5))
    plt.plot(acf_values, marker='o', markersize=2, linestyle='-', color='b')
    plt.axvline(x=24, color='r', linestyle='--', alpha=0.6, label='24h (Daily)')
    plt.axvline(x=168, color='g', linestyle='--', alpha=0.6, label='168h (Weekly)')
    
    plt.title('Autocorrelation of User Events (Temporal Pattern Detection)')
    plt.xlabel('Lag (Hours)')
    plt.ylabel('Autocorrelation Coefficient')
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.tight_layout()
    
    # Save the plot to a file so it can be viewed
    plt.savefig('autocorrelation_plot.png')
    print("\n📊 Plot saved as 'autocorrelation_plot.png'")

if __name__ == "__main__":
    main()
