# Temporal Pattern Detection POC
# --------------------------------
# Goal:
# 1. Generate synthetic event data (daily call to mom ~6pm with noise, weekly Saturday 10am call)
# 2. Convert to time-binned binary time series
# 3. Compute autocorrelation for daily and weekly lags
# 4. Show detected periodicity

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# -------------------------------
# 1. Generate Synthetic Data
# -------------------------------

np.random.seed(42)

days = 30
bins_per_day = 24  # 1-hour bins for simplicity
total_bins = days * bins_per_day

# Create empty binary time series
time_series = np.zeros(total_bins)

# Daily pattern: call mom around 6pm ± 1 hour (bin 18)
for d in range(days):
    if np.random.rand() > 0.1:  # 10% chance of missing the day
        hour = 18 + np.random.choice([-1, 0, 1])
        idx = d * bins_per_day + hour
        time_series[idx] = 1

# Weekly pattern: Saturday 10am call (every 7 days, bin 10)
for d in range(0, days, 7):
    hour = 10
    idx = d * bins_per_day + hour
    time_series[idx] = 1

# -------------------------------
# 2. Autocorrelation Function
# -------------------------------

def autocorrelation(series, lag):
    return np.sum(series[lag:] * series[:-lag])

# Compute autocorrelation for lags up to 14 days
lags = np.arange(1, 14 * bins_per_day)
acf_values = [autocorrelation(time_series, lag) for lag in lags]

# -------------------------------
# 3. Detect Daily & Weekly Peaks
# -------------------------------

daily_lag = bins_per_day
weekly_lag = 7 * bins_per_day

daily_score = autocorrelation(time_series, daily_lag)
weekly_score = autocorrelation(time_series, weekly_lag)

print("Daily Lag Score (24h):", daily_score)
print("Weekly Lag Score (7 days):", weekly_score)

# -------------------------------
# 4. Plot Autocorrelation
# -------------------------------

plt.figure()
plt.plot(lags, acf_values)
plt.xlabel("Lag (hours)")
plt.ylabel("Autocorrelation Score")
plt.title("Autocorrelation of Synthetic Call Events")
plt.show()

print("POC complete: Peaks near 24h and 168h indicate daily and weekly patterns.")
